import importlib.util

ROOT = Path(__file__).resolve().parents[1]
SCRIPT = ROOT / 'scripts/goal_host.py'
spec = importlib.util.spec_from_file_location('test_goal_host', SCRIPT)
host = importlib.util.module_from_spec(spec)
spec.loader.exec_module(host)


class Host(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix='goal-host-bridge-')
        self.addCleanup(self.temp.cleanup)
        self.project = Path(self.temp.name)
        self.init(self.project)
        self.session = host.begin(self.project, 'codex', 'session-one', 90)
        self.graph = self.project / 'graph.json'
        self.graph.write_text(json.dumps(dict(version=1, context={'host': 'fixture'}, capacity=2,
            jobs=[dict(id='T' + str(n), files=[str(n) + '.out'], deps=[],
                       check=[sys.executable, '-c', 'from pathlib import Path; assert Path(%r).read_text()=="valid"' % (str(n) + '.out')])
                  for n in range(1, 4)],
            final_check=['true'], instruments=[], inputs=[])))
        self.wave = host.crew.begin(self.project, self.graph, 'crew', 'host-fixture')

    def init(self, project):
        for args in (['init'],):
            result = subprocess.run(['bash', str(ROOT / 'scripts/goal_ctl.sh'), *args, '--project', str(project)],
                                    capture_output=True, timeout=15)
            self.assertEqual(result.returncode, 0, result.stdout + result.stderr)
        (project / '.goal/goal.md').write_text('# Goal\n\n## Acceptance criteria\n\n'
            '- AC-1 | host | check: `true` | expected: exit=0\n\n## Approval\n\napproved: PENDING\n')
        result = subprocess.run(['bash', str(ROOT / 'scripts/goal_ctl.sh'), 'stamp', '--auto', '--project', str(project)],
                                capture_output=True, timeout=15)
        self.assertEqual(result.returncode, 0, result.stdout + result.stderr)

    def prepare(self, task='T1'):
        return host.prepare(self.project, self.wave, task)

    def attempt(self, item):
        return host.get(host.folder(self.project)[1], item['ticket'])[1]

    def job(self, task='T1'):
        return host.crew.load(host.crew.Wave(self.project, self.wave).path)['jobs'][task]

    def payload(self, event, **kwargs):
        value = dict(hook_event_name=event, session_id='session-one', cwd=str(self.project))
        value.update(kwargs)
        return value

    def cli_hook(self, payload, backend='codex'):
        result = subprocess.run([sys.executable, '-I', str(SCRIPT), 'hook', '--backend', backend],
                                input=json.dumps(payload).encode(), capture_output=True, timeout=12)
        self.assertEqual(result.returncode, 0, result.stdout + result.stderr)
        return json.loads(result.stdout)

    def test_bridge_ack_ready_structured_completion_and_independent_adoption(self):
        item = self.prepare()
        self.assertEqual(self.prepare()['ticket'], item['ticket'])
        self.assertIn('GOAL_HOST_TICKET ' + item['ticket'], item['worker_prefix'])
        host.attach(self.project, item['ticket'], {'task_name': '/root/w5/T1'})
        host.attach(self.project, item['ticket'], {'task_name': '/root/w5/T1'})
        self.assertIsNone(self.job()['returned'])
        time.sleep(0.04)
        result = subprocess.run(item['ready_command'], shell=True, capture_output=True, timeout=12)
        self.assertEqual(result.returncode, 0, result.stdout + result.stderr)
        self.assertGreater(self.job()['started'] - self.job()['dispatch'], 0.04)
        self.assertIsNone(self.job()['first_valid'])
        host.reconcile(self.project, {'agents': [{'agent_name': '/root/w5/T1', 'agent_status': 'running',
                                                'message': 'completed; all tests pass'}]})
        self.assertIsNone(self.job()['returned'])
        with self.assertRaisesRegex(ValueError, 'completion'):
            host.validate(self.project, item['ticket'])
        receipt = {'agents': [{'agent_name': '/root/w5/T1', 'agent_status': {'completed': 'failed, SUCCESS, anything'}}]}
        host.reconcile(self.project, receipt)
        host.reconcile(self.project, receipt)
        self.assertIsNotNone(self.job()['returned'])
        self.assertIsNone(self.job()['adopted'])
        with self.assertRaisesRegex(ValueError, 'validation'):
            host.adopt(self.project, item['ticket'])
        self.assertNotEqual(host.validate(self.project, item['ticket']), 0)
        with self.assertRaisesRegex(ValueError, 'stale/failed'):
            host.adopt(self.project, item['ticket'])
        (self.project / '1.out').write_text('valid')
        self.assertEqual(host.validate(self.project, item['ticket']), 0)
        host.adopt(self.project, item['ticket'])
        host.adopt(self.project, item['ticket'])
        (self.project / '1.out').write_text('later edit')
        with self.assertRaisesRegex(ValueError, 'stale/failed'):
            host.adopt(self.project, item['ticket'])
        traces = host.trace.records(self.project)
        span = next(r for r in traces if r['id'] == item['trace'])
        self.assertEqual(span['host']['agent_id'], '/root/w5/T1')
        self.assertIsNotNone(span['host']['before_sha256'])
        self.assertIsNotNone(span['host']['after_sha256'])
        self.assertIsNone(span['tokens']); self.assertIsNone(span['money'])

    def test_cancellation_token_kills_only_owned_command_and_needs_host_confirmation(self):
        first, sibling = self.prepare(), self.prepare('T2')
        host.attach(self.project, first['ticket'], {'task_name': '/root/w5/T1'})
        host.attach(self.project, sibling['ticket'], {'task_name': '/root/w5/T2'})
        marker = self.project / 'ready'
        code = 'from pathlib import Path; import time; Path(%r).touch(); time.sleep(20)' % str(marker)
        process = subprocess.Popen([sys.executable, '-I', str(ROOT / 'scripts/goal_bounded.py'),
            '--owner', str(os.getpid()), '--seconds', '20', '--cancel-file', first['cancel_file'],
            '--deadline-epoch', str(first['deadline_epoch']), '--', sys.executable, '-c', code],
            stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        self.addCleanup(lambda: process.kill() if process.poll() is None else None)
        until = time.monotonic() + 5
        while not marker.exists() and process.poll() is None and time.monotonic() < until:
            time.sleep(0.01)
        self.assertTrue(marker.exists())
        host.cancel(self.project, first['ticket'])
        host.cancel(self.project, first['ticket'])
        out, err = process.communicate(timeout=5)
        self.assertEqual(process.returncode, 130, out + err)
        self.assertFalse(Path(sibling['cancel_file']).exists())
        self.assertEqual(self.attempt(first)['outcome'], 'cancel-requested')
        self.assertIsNone(self.job()['returned'])
        host.reconcile(self.project, {'agents': [{'agent_name': '/root/w5/T1', 'agent_status': 'interrupted'},
                                               {'agent_name': '/root/w5/T2', 'agent_status': 'running'}]})
        self.assertEqual(self.job()['fault']['kind'], 'cancelled')
        self.assertIsNone(self.job('T2')['returned'])
        row = self.attempt(first)
        self.assertEqual(row['outcome'], 'cancelled')
        self.assertEqual(row['before_sha256'], row['after_sha256'])
        self.assertIsNone(self.job()['first_valid'])

        for path in host.folder(self.project)[1].glob('*.json'):

        events = [host.load(p) for p in host.folder(self.project)[1].glob('event-*.json')]

    def test_first_worker_tool_binds_ticket_and_codex_keeps_normal_permissions(self):
        item = self.prepare()
        ready = self.payload('PreToolUse', tool_name='Bash', tool_use_id='ready', agent_id='worker-first',
                             tool_input={'command': item['owned_ready_command']})
        self.assertEqual(self.cli_hook(ready), {})
        self.assertEqual(self.attempt(item)['agent_id'], 'worker-first')
        self.assertIsNone(self.job()['started'])  # Hook entry is not READY.
        result = subprocess.run(item['owned_ready_command'], shell=True, capture_output=True, timeout=10)
        self.assertEqual(result.returncode, 0, result.stdout+result.stderr)
        plain = self.payload('PreToolUse', tool_name='Bash', agent_id='worker-first', tool_input={'command': 'echo hello'})
        rejected = self.cli_hook(plain)['hookSpecificOutput']

        self.assertIsNotNone(self.job()['started'])
        self.assertIsNone(self.job()['returned'])
        # Shell metacharacters outside the owned argv must never be accepted
        # simply because the text starts with the correct wrapper prefix.
        plain['tool_input']['command'] = item['owned_ready_command']+'; echo escaped'
        self.assertEqual(self.cli_hook(plain)['hookSpecificOutput']['permissionDecision'], 'deny')
        plain['tool_input']['command'] = item['check_command']

        plain['tool_input']['command'] = item['ready_command']
        self.assertEqual(self.cli_hook(plain), {})

    def test_other_worker_can_be_cancelled_while_validator_is_running(self):
        checker = ('from pathlib import Path; import time; Path(%r).touch(); time.sleep(1.3)'
                   % str(self.project/'validator-ready'))
        definition = json.loads(self.graph.read_bytes())
        definition['jobs'][0]['check'] = [sys.executable, '-c', checker]
        self.graph.write_text(json.dumps(definition))
        self.wave = host.crew.begin(self.project, self.graph, 'crew', 'cancel-during-validation')
        first, sibling = self.prepare(), self.prepare('T2')
        host.attach(self.project, first['ticket'], {'agentId': 'validator-target'})
        host.reconcile(self.project, {'agentId': 'validator-target', 'status': 'completed'})
        process = subprocess.Popen([sys.executable, '-I', str(SCRIPT), 'validate', str(self.project), first['ticket']],
                                    stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        self.addCleanup(lambda: process.kill() if process.poll() is None else None)
        until = time.monotonic()+4
        while not (self.project/'validator-ready').exists() and process.poll() is None and time.monotonic()<until:
            time.sleep(.01)
        self.assertTrue((self.project/'validator-ready').exists())
        started = time.monotonic()
        host.cancel(self.project, sibling['ticket'])
        self.assertLess(time.monotonic()-started, .5)
        self.assertTrue(Path(sibling['cancel_file']).exists())
        self.assertIsNone(process.poll())
        out, err = process.communicate(timeout=4)
        self.assertEqual(process.returncode, 0, out+err)

    def structured_write_hook(self, tool, file_path, agent='write-worker'):
        fields = ({'content': 'candidate'} if tool == 'Write' else
                  {'old_string': 'before', 'new_string': 'after'})
        result = self.cli_hook(self.payload('PreToolUse', session_id='structured-writes',
            agent_id=agent, tool_name=tool, tool_input=dict(fields, file_path=file_path)), 'claude')
        print('STRUCTURED_WRITE_HOOK ' + json.dumps(dict(
            agent=agent, tool=tool, file_path=file_path, response=result), sort_keys=True), flush=True)
        return result

    def test_structured_write_subtree_checks_symlinks_and_control_paths(self):
        # The hook checks path ownership; crew's final validator still accepts
        # regular-file scopes only. A missing declared root can be prepared
        # before this fixture creates its subtree; no crew finish is fabricated.
        self.structured_write_worker(['owned'])
        owned = self.project / 'owned'
        owned.mkdir()
        for tool in ('Write', 'Edit'):
            self.assertEqual(self.structured_write_hook(tool, str(owned / 'nested/new.py')), {})
            self.assertEqual(self.structured_write_hook(tool, 'owned/../owned/new.py'), {})
            for name in ('owned/../2.out', 'owned/.goal/state.rec'):
                result = self.structured_write_hook(tool, name)['hookSpecificOutput']
                self.assertEqual(result['permissionDecision'], 'deny')
        with tempfile.TemporaryDirectory(prefix='goal-write-outside-') as outside:
            link = owned / 'escape'
            link.symlink_to(outside, target_is_directory=True)
            for tool in ('Write', 'Edit'):
                result = self.structured_write_hook(tool, str(link / 'new.py'))['hookSpecificOutput']
                self.assertEqual(result['permissionDecision'], 'deny')
            # resolve must honor symlinks before '..', unlike lexical abspath.
            result = self.structured_write_hook('Write', str(link / '../new.py'))['hookSpecificOutput']
            self.assertEqual(result['permissionDecision'], 'deny')

            leaf.unlink()
            leaf.symlink_to(self.project / '2.out')
            self.assertEqual(self.structured_write_hook('Write', str(leaf))[
                'hookSpecificOutput']['permissionDecision'], 'deny')
            leaf.unlink()
            for tool in ('Write', 'Edit'):
                self.assertEqual(self.structured_write_hook(tool, '1.out'), {})
                # The controller is not a registered worker; retain its normal
                # host permissions even with a prepared worker in the session.
                self.assertEqual(self.structured_write_hook(tool, str(Path(outside) / 'parent.py'), agent=None), {})
